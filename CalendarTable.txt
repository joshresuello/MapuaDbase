Calendar = 
VAR StartDate = DATE(2020, 1, 1)
VAR EndDate = DATE(2030, 12, 31)

RETURN
ADDCOLUMNS(
    CALENDAR(StartDate, EndDate),

    -- Date Keys
    "DateKey", FORMAT([Date], "YYYYMMDD"),

    -- Year
    "Year", YEAR([Date]),

    -- Month
    "Month Number", MONTH([Date]),
    "Month Name", FORMAT([Date], "MMMM"),
    "Month Short", FORMAT([Date], "MMM"),

    -- Quarter
    "Quarter Number", QUARTER([Date]),
    "Quarter", "Q" & QUARTER([Date]),
    "Year Quarter", YEAR([Date]) & " Q" & QUARTER([Date]),

    -- Year-Month
    "Year Month", FORMAT([Date], "YYYY-MM"),
    "Year Month Number", YEAR([Date]) * 100 + MONTH([Date]),

    -- Week
    "Week Number", WEEKNUM([Date], 2),
    "Week", "Week " & WEEKNUM([Date], 2),

    -- Day
    "Day", DAY([Date]),
    "Day Name", FORMAT([Date], "DDDD"),
    "Day Short", FORMAT([Date], "DDD"),
    "Day of Week", WEEKDAY([Date], 2),

    -- Working Days
    "Is Weekend",
        IF(
            WEEKDAY([Date], 2) >= 6,
            TRUE(),
            FALSE()
        ),

    "Is Working Day",
        IF(
            WEEKDAY([Date], 2) <= 5,
            TRUE(),
            FALSE()
        ),

    -- Month Boundaries
    "Month Start",
        DATE(
            YEAR([Date]),
            MONTH([Date]),
            1
        ),

    "Month End",
        EOMONTH([Date], 0),

    -- Quarter Boundaries
    "Quarter Start",
        DATE(
            YEAR([Date]),
            (QUARTER([Date]) - 1) * 3 + 1,
            1
        ),

    "Quarter End",
        EOMONTH(
            DATE(
                YEAR([Date]),
                QUARTER([Date]) * 3,
                1
            ),
            0
        ),

    -- Year Boundaries
    "Year Start",
        DATE(
            YEAR([Date]),
            1,
            1
        ),

    "Year End",
        DATE(
            YEAR([Date]),
            12,
            31
        ),
    /* Dev note: Adjustment dependent sa fiscal calendar ng organization. 
        This code is specific para sa fiscal calendar starting April 1 */

        
    -- Fiscal Calendar (April Start)
    "Fiscal Year",
        IF(
            MONTH([Date]) >= 4,
            YEAR([Date]) + 1,
            YEAR([Date])
        ),

    "Fiscal Month Number",
        MOD(
            MONTH([Date]) - 4,
            12
        ) + 1,

    "Fiscal Month",
        "Month " &
        (
            MOD(
                MONTH([Date]) - 4,
                12
            ) + 1
        ),

    "Fiscal Quarter Number",
        INT(
            MOD(
                MONTH([Date]) - 4,
                12
            ) / 3
        ) + 1,

    "Fiscal Quarter",
        "Q" &
        (
            INT(
                MOD(
                    MONTH([Date]) - 4,
                    12
                ) / 3
            ) + 1
        ),

    -- Current Period Flags
    "Is Today",
        [Date] = TODAY(),

    "Is Current Year",
        YEAR([Date]) = YEAR(TODAY()),

    "Is Current Month",
        YEAR([Date]) = YEAR(TODAY())
            && MONTH([Date]) = MONTH(TODAY()),

    "Is Current Quarter",
        YEAR([Date]) = YEAR(TODAY())
            && QUARTER([Date]) = QUARTER(TODAY())
)
